/** Every visible control clipped by an overflow:hidden/clip ancestor or the viewport. */
async function clipped(page: Page) {
  return page.evaluate(() => {
    const out: string[] = [];
    const els = document.querySelectorAll<HTMLElement>("button, a[href], [role=button], input, select");
    for (const el of els) {
      const r = el.getBoundingClientRect();
      if (r.width === 0 || r.height === 0) continue;
      const cs = getComputedStyle(el);
      if (cs.visibility === "hidden") continue;
      let p = el.parentElement;
      let reason = "";
      while (p && p !== document.body) {
        const s = getComputedStyle(p);
        const ox = s.overflowX, oy = s.overflowY;
        if (["auto","scroll"].includes(ox) || ["auto","scroll"].includes(oy)) break;
        if (ox !== "visible" || oy !== "visible") {
          const pr = p.getBoundingClientRect();
          const clipY = (oy === "hidden" || oy === "clip");
          const clipX = (ox === "hidden" || ox === "clip");
          if (clipY && (r.bottom > pr.bottom + 1 || r.top < pr.top - 1)) { reason = `clippedY by ${p.tagName}.${(p.className||"").toString().slice(0,60)}`; break; }
          if (clipX && (r.right > pr.right + 1 || r.left < pr.left - 1)) { reason = `clippedX by ${p.tagName}.${(p.className||"").toString().slice(0,60)}`; break; }
        }
        p = p.parentElement;
      }
      if (!reason && (r.left < -1 || r.right > innerWidth + 1)) reason = "outside viewport X";
      if (reason) out.push(`${(el.textContent || el.getAttribute("aria-label") || el.tagName).trim().slice(0, 40)} :: ${reason}`);
    }
    return [...new Set(out)];
  });
}

async function snap(page: Page, name: string, report: Record<string, string[]>) {
  await page.waitForTimeout(800);
  await page.screenshot({ path: `${OUT}/${test.info().project.name}-${name}.png` });
  report[name] = await clipped(page);
}
